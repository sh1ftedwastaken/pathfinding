cmake_minimum_required(VERSION 3.18)

project(pathfinding_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# By default, build every Python version listed below. Set
# PATHFINDING_PYTHON_VERSION to build only one version instead.
set(PATHFINDING_PYTHON_VERSION "" CACHE STRING
    "Exact Python version (e.g. 3.12); empty = build all listed versions")
set(PATHFINDING_PYTHON_EXECUTABLE "" CACHE FILEPATH
    "Python interpreter path for a single-version build; empty = auto-detect")
set(PATHFINDING_DIST_DIR "${CMAKE_SOURCE_DIR}/dist" CACHE PATH
    "Directory where version-tagged Python extension files are collected")
set(PATHFINDING_PYTHON_VERSIONS "3.12;3.13;3.14;3.15" CACHE STRING
    "Python versions to build")
option(PATHFINDING_BUILD_ALL_PYTHON
    "Build every version in PATHFINDING_PYTHON_VERSIONS"
    ON)

if(PATHFINDING_BUILD_ALL_PYTHON AND NOT PATHFINDING_PYTHON_VERSION)
    set(_python_targets)

    foreach(_version IN LISTS PATHFINDING_PYTHON_VERSIONS)
        string(REPLACE "." "_" _tag "${_version}")
        set(_build_dir "${CMAKE_BINARY_DIR}/py${_tag}")

        add_custom_target(build_python_${_tag}
            COMMAND "${CMAKE_COMMAND}"
                -S "${CMAKE_SOURCE_DIR}"
                -B "${_build_dir}"
                -G "${CMAKE_GENERATOR}"
                "-DPATHFINDING_BUILD_ALL_PYTHON=OFF"
                "-DPATHFINDING_PYTHON_VERSION=${_version}"
                "-DPATHFINDING_DIST_DIR=${PATHFINDING_DIST_DIR}"
            COMMAND "${CMAKE_COMMAND}" --build "${_build_dir}" --config Release
            COMMENT "Building pathfinding for Python ${_version}"
            USES_TERMINAL
            VERBATIM
        )

        list(APPEND _python_targets build_python_${_tag})
    endforeach()

    add_custom_target(build_all_python ALL DEPENDS ${_python_targets})
    return()
endif()

add_library(pathfinding_core STATIC
    src/AStart.cpp
    src/Dijkstra.cpp
    src/VoxelGrid.cpp
)

target_include_directories(pathfinding_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(pathfinding_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(PATHFINDING_PYTHON_EXECUTABLE)
    set(Python_EXECUTABLE "${PATHFINDING_PYTHON_EXECUTABLE}"
        CACHE FILEPATH "Python interpreter" FORCE)
endif()

if(PATHFINDING_PYTHON_VERSION)
    find_package(Python ${PATHFINDING_PYTHON_VERSION} EXACT REQUIRED
                 COMPONENTS Interpreter Development.Module)
else()
    find_package(Python 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
endif()

message(STATUS "Python ${Python_VERSION}: ${Python_EXECUTABLE}")

set(PYBIND11_FINDPYTHON ON CACHE STRING "Use CMake's FindPython")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/pybind11/CMakeLists.txt")
    add_subdirectory(external/pybind11)
else()
    find_package(pybind11 CONFIG REQUIRED)
endif()

pybind11_add_module(pathfinding src/binding.cpp)
target_link_libraries(pathfinding PRIVATE pathfinding_core)

# MinGW: statically link the compiler runtime so the .pyd needs only Python
# and normal system DLLs. MSVC and other toolchains keep their normal runtime.
if(MINGW)
    target_link_options(pathfinding PRIVATE -static)
endif()

add_custom_command(TARGET pathfinding POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${PATHFINDING_DIST_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "$<TARGET_FILE:pathfinding>"
        "${PATHFINDING_DIST_DIR}/$<TARGET_FILE_NAME:pathfinding>"
    COMMENT "Copying $<TARGET_FILE_NAME:pathfinding> to ${PATHFINDING_DIST_DIR}"
    VERBATIM
)

install(TARGETS pathfinding
    RUNTIME DESTINATION .
    LIBRARY DESTINATION .
)
